def parse_lastname(author: str) -> str:
    '''Extrait le nom de famille d un auteur formate "Prenom Nom" ou "Nom, Prenom".'''
    s = author.strip()
    if ',' in s:
        return s.split(',')[0].strip()
    parts = s.split()
    return parts[-1] if parts else s
