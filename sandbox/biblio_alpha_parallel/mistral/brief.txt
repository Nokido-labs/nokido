=== FICHIER 1 : app/forge_biblio_sanitizer.py ===

Module avec ces fonctions publiques :

def validate_doi(s: str) -> bool:
    Regex case-insensitive: ^10\.\d{4,9}/[-._;()/:A-Z0-9]+$

def validate_arxiv_id(s: str) -> bool:
    Regex: ^\d{4}\.\d{4,5}(v\d+)?$

def validate_isbn(s: str) -> bool:
    ISBN-10 (10 digits + checksum mod 11) ou ISBN-13 (13 digits + checksum mod 10)

def is_url_whitelisted(url: str) -> tuple[bool, str]:
    WHITELIST domaines acceptes (return True, ''):
    arxiv.org, doi.org, oadoi.org, openalex.org, openlibrary.org,
    semanticscholar.org, scholar.google.com, openreview.net, neurips.cc,
    aclanthology.org, ijcai.org, acm.org, ieee.org, springer.com,
    sciencedirect.com, github.com, zenodo.org, figshare.com, osf.io,
    et tout *.edu et *.gov
    BLACKLIST (return False, 'url_blacklist'):
    medium.com, dev.to, researchgate.net, academia.edu
    HTTP non chiffre -> (False, 'http_not_allowed')
    URL malforme -> (False, 'url_invalid')

def sanitize_content(text: str) -> tuple[bool, str]:
    Detection patterns OWASP LLM01 (instruction override):
    Regex: \b(ignore|disregard|override|you are now|act as|from now on)\b.*(previous|instructions|rules)
    EXCEPTION : si encadre dans triple backtick ou commentaire HTML <!-- example -->
    Returns (True, '') si OK, (False, 'prompt_injection') sinon.

def sanitize_entry(entry: dict) -> tuple[bool, str]:
    Combine validations:
    - 'title' obligatoire, length >= 5 -> sinon 'missing_required_field'
    - 'triggered_by_idea_id' obligatoire -> sinon 'missing_required_field'
    - 'source_kind' in ['human_paste','agent_extract','agent_research'] -> sinon 'invalid_source_kind'
    - 'type' in ['book','paper','url','concept'] -> sinon 'invalid_type'
    - 'doi' si present -> validate_doi -> sinon 'doi_invalid'
    - 'url' si present -> is_url_whitelisted -> sinon retourne sa raison
    - 'pdf_url' si present -> is_url_whitelisted -> sinon retourne sa raison
    - 'description' si present -> sanitize_content -> sinon 'prompt_injection'
    Returns (True, '') si tout OK.
